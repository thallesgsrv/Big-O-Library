{{- /* Tabelas do Markdown. Igual ao padrão do Hugo, com dois ajustes de acessibilidade: 1) cabeçalho totalmente vazio (tabelas "chave e valor", em que o Markdown exige uma linha de título) vira uma tabela sem , em vez de uma linha de cabeçalho vazia; 2) célula de cabeçalho vazia (o canto de uma tabela comparativa) recebe o texto "Item", visível só para leitores de tela, para que toda coluna tenha um nome. */ -}} {{- $vazio := true -}} {{- range .THead }}{{ range . }}{{ if strings.TrimSpace (printf "%s" .Text) }}{{ $vazio = false }}{{ end }}{{ end }}{{ end -}} {{- if not $vazio }} {{- range .THead }} {{- range . }} {{ if strings.TrimSpace (printf "%s" .Text) }}{{ .Text }}{{ else }}Item{{ end }} {{- end }} {{- end }} {{- end }} {{- range .TBody }} {{- range . }} {{ .Text }} {{- end }} {{- end }}
