{{- /* Scripts do núcleo (menu, tema, sidebar, busca...) unidos em js/main.js. Cada arquivo roda isolado (IIFE + try/catch), então um erro não derruba os demais. */ -}} {{- $parts := slice -}} {{- range resources.Match "js/core/*.js" -}} {{- $res := resources.ExecuteAsTemplate .Name $ . -}} {{- $parts = $parts | append (printf "(function(){try{\n%s\n}catch(err){console.error('[big-o-library] erro em %s:',err)}})();" $res.Content .Name) -}} {{- end -}} {{- $scripts := resources.FromString "js/main.js" (delimit $parts "\n") -}} {{- if hugo.IsProduction -}} {{- $scripts = $scripts | minify | fingerprint -}} {{- end -}}
